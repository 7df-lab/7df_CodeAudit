### 1.2 Go 代码规范

#### 1.2.1 代码风格

```go
// ✓ 正确示例
package service

import (
    "context"
    "fmt"

    "github.com/codeaudit/internal/model"
    "github.com/codeaudit/internal/repo"
)

// UserService 用户服务接口
type UserService interface {
    GetUser(ctx context.Context, id string) (*model.User, error)
    CreateUser(ctx context.Context, user *model.User) error
}

// userService 用户服务实现
type userService struct {
    userRepo repo.UserRepo
    cache    CacheService
}

// NewUserService 创建用户服务实例
func NewUserService(userRepo repo.UserRepo, cache CacheService) UserService {
    return &userService{
        userRepo: userRepo,
        cache:    cache,
    }
}

// GetUser 获取用户信息
func (s *userService) GetUser(ctx context.Context, id string) (*model.User, error) {
    // 先尝试从缓存获取
    if user, err := s.cache.GetUser(ctx, id); err == nil {
        return user, nil
    }

    // 从数据库获取
    user, err := s.userRepo.GetByID(ctx, id)
    if err != nil {
        return nil, fmt.Errorf("failed to get user %s: %w", id, err)
    }

    // 写入缓存
    if err := s.cache.SetUser(ctx, id, user); err != nil {
        // 缓存失败不影响主流程，只记录日志
        log.Warnf("failed to cache user %s: %v", id, err)
    }

    return user, nil
}
```

```go
// ✓ 正确的错误处理
func (s *service) Process(ctx context.Context, input *Input) (*Output, error) {
    if err := input.Validate(); err != nil {
        return nil, fmt.Errorf("invalid input: %w", err)
    }

    result, err := s.repo.Save(ctx, input)
    if err != nil {
        return nil, fmt.Errorf("failed to save: %w", err)
    }

    return result, nil
}

// ✗ 错误的错误处理
func (s *service) Process(ctx context.Context, input *Input) (*Output, error) {
    result, err := s.repo.Save(ctx, input)
    if err != nil {
        return nil, err  // 不要直接返回原始错误
    }
    return result, nil
}
```

#### 1.2.1 代码风格

```python
# ✓ 正确示例
"""用户服务模块"""

from typing import Optional, List
from dataclasses import dataclass
import logging

from codeaudit.models.user import User
from codeaudit.repositories.user_repo import UserRepository

logger = logging.getLogger(__name__)

@dataclass
class UserServiceConfig:
    """用户服务配置"""
    cache_ttl: int = 3600
    max_retries: int = 3

class UserService:
    """用户服务类"""

    def __init__(
        self,
        user_repo: UserRepository,
        config: Optional[UserServiceConfig] = None
    ):
        self._user_repo = user_repo
        self._config = config or UserServiceConfig()

    async def get_user(self, user_id: str) -> Optional[User]:
        """
        获取用户信息

        Args:
            user_id: 用户ID

        Returns:
            用户对象，如果不存在返回None

        Raises:
            ValueError: 当user_id为空时
            DatabaseError: 当数据库查询失败时
        """
        if not user_id:
            raise ValueError("user_id cannot be empty")

        try:
            user = await self._user_repo.get_by_id(user_id)
            return user
        except Exception as e:
            logger.error(f"Failed to get user {user_id}: {e}")
            raise
```

#### 1.3.1 代码风格

```typescript
// ✓ 正确示例
import React, { useState, useEffect, useCallback } from 'react';
import { Card, Table, Button, message } from 'antd';
import { Finding, FindingFilter } from '@/types/finding';
import { getFindings } from '@/services/finding';

interface FindingListProps {
  taskId: string;
  onSelect?: (finding: Finding) => void;
}

const FindingList: React.FC<FindingListProps> = ({ taskId, onSelect }) => {
  const [findings, setFindings] = useState<Finding[]>([]);
  const [loading, setLoading] = useState<boolean>(false);
  const [filter, setFilter] = useState<FindingFilter>({});

  const fetchFindings = useCallback(async () => {
    setLoading(true);
    try {
      const data = await getFindings(taskId, filter);
      setFindings(data);
    } catch (error) {
      message.error('Failed to fetch findings');
      console.error(error);
    } finally {
      setLoading(false);
    }
  }, [taskId, filter]);

  useEffect(() => {
    fetchFindings();
  }, [fetchFindings]);

  const handleSelect = (record: Finding) => {
    onSelect?.(record);
  };

  return (
    <Card title="Audit Findings">
      <Table
        dataSource={findings}
        loading={loading}
        rowKey="id"
        onRow={(record) => ({
          onClick: () => handleSelect(record),
        })}
      >
        <Table.Column title="ID" dataIndex="findingId" />
        <Table.Column title="Severity" dataIndex="severity" />
        <Table.Column title="CWE" dataIndex="cweId" />
        <Table.Column title="File" dataIndex="filePath" />
        <Table.Column title="Confidence" dataIndex="confidence" />
      </Table>
    </Card>
  );
};

export default FindingList;
```

### 5.2 gRPC API 规范

> **S1 规则**：数据结构唯一事实源为 `codeaudit_common.proto`（V2.0），本文档不内嵌 proto 定义（V1.x 此处内嵌的 ProjectService 示例与 V2.0 冲突——旧示例 `Project.id` 应为 `project_id`、`scan_mode` 应为 `ScanMode` 枚举而非 string——已删除）。

开发约定：

1. 服务端实现 proto 中的 service 时，一个部署服务可承载多个 logical service（映射见《01 总体架构设计》§4.2），但每个 logical service 必须独立实现类、独立配置；
2. 全部写操作 RPC 必须读取 `RequestMetadata` 幂等键（缺失返回 INVALID_ARGUMENT，见《03 接口规范》§2）；
3. 错误返回统一使用 gRPC 标准码 + proto `ErrorResponse` 结构（错误码表见《03 接口规范》§3）；
4. 新增字段只增不改号，新增枚举只追加——破坏性变更走 `.v2` 包双活流程（见《03 接口规范》§6）。

```go
    //   - *User: 用户对象
    //   - error: 错误信息，如果用户不存在返回 ErrUserNotFound
    GetUser(ctx context.Context, id string) (*User, error)
}
```
