## 1. 测试概述

### 1.1 文档目的

本测试计划定义了CodeAudit智能代码安全审计系统的测试策略、测试范围、测试方法和验收标准，确保系统质量满足预期目标。

### 1.2 测试目标

| 目标 | 具体指标 |
|------|---------|
| 功能完整性 | 所有需求功能100%实现 |
| 功能正确性 | 核心功能测试通过率100% |
| 性能达标 | 满足性能验收标准 |
| 安全合规 | 无高危安全漏洞 |
| 兼容性 | 支持主流浏览器和操作系统 |

# Python 单元测试示例
class TestVulnerabilityDetector:
    """漏洞检测器测试"""

    @pytest.fixture
    def detector(self):
        return VulnerabilityDetector()

    def test_detect_sql_injection(self, detector):
        """测试SQL注入检测"""
        code = '''
        def get_user(user_id):
            query = f"SELECT * FROM users WHERE id = {user_id}"
            return db.execute(query)
        '''

        result = detector.detect(code)
        assert len(result) == 1
        assert result[0].cwe_id == "CWE-89"
        assert result[0].confidence > 0.8

    def test_no_vulnerability_with_parameterized_query(self, detector):
        """测试参数化查询不报漏洞"""
        code = '''
        def get_user(user_id):
            query = "SELECT * FROM users WHERE id = %s"
            return db.execute(query, (user_id,))
        '''

        result = detector.detect(code)
        assert len(result) == 0

    def test_detect_xss(self, detector):
        """测试XSS检测"""
        code = '''
        def render_page(user_input):
            return f"<div>{user_input}</div>"
        '''

        result = detector.detect(code)
        assert any(r.cwe_id == "CWE-79" for r in result)

```javascript
// k6 性能测试脚本
import http from 'k6/http';
import { check, sleep } from 'k6';
import { Rate, Trend } from 'k6/metrics';

// 自定义指标
const errorRate = new Rate('errors');
const apiLatency = new Trend('api_latency');

// 测试配置
export const options = {
  stages: [
    { duration: '2m', target: 50 },   // 逐渐增加到50并发
    { duration: '5m', target: 50 },   // 保持50并发
    { duration: '2m', target: 100 },  // 增加到100并发
    { duration: '5m', target: 100 },  // 保持100并发
    { duration: '2m', target: 0 },    // 逐渐降到0
  ],
  thresholds: {
    'http_req_duration': ['p(95)<200'],  // 95%请求<200ms
    'errors': ['rate<0.01'],              // 错误率<1%
  },
};

// 测试场景
export default function () {
  const BASE_URL = __ENV.API_URL || 'http://localhost:8080';

  // 登录获取token
  const loginRes = http.post(`${BASE_URL}/v1/auth/login`, JSON.stringify({
    username: 'testuser',
    password: 'testpass',
  }), {
    headers: { 'Content-Type': 'application/json' },
  });

  check(loginRes, {
    'login successful': (r) => r.status === 200,
  });

  const token = loginRes.json('data.token');
  const headers = {
    'Authorization': `Bearer ${token}`,
    'Content-Type': 'application/json',
  };

  // 获取项目列表
  const projectsRes = http.get(`${BASE_URL}/v1/projects`, { headers });

  const success = check(projectsRes, {
    'projects status 200': (r) => r.status === 200,
    'projects response time < 200ms': (r) => r.timings.duration < 200,
  });

  errorRate.add(!success);
  apiLatency.add(projectsRes.timings.duration);

  sleep(1);
}
```

### 5.5 AI准确性测试

#### 5.5.1 测试基准

| 基准数据集 | 规模 | 评估指标 |
|-----------|------|---------|
| DiverseVul | 367,752个函数 | F1, Precision, Recall |
| BigVul | 406,373个函数 | F1, Precision, Recall |
| PrimeVul | 高质量标注 | F1, Precision, Recall |
| 自建测试集 | 1000个漏洞 | F1, Precision, Recall |

#### 5.5.2 评估指标

```python
# AI准确性评估脚本
class AIAccuracyEvaluator:
    """AI准确性评估器"""

    def evaluate(self, predictions: List[Finding], ground_truth: List[Finding]) -> dict:
        """评估AI检测准确性"""
        tp = 0  # 真阳性
        fp = 0  # 假阳性
        fn = 0  # 假阴性

        matched_gt = set()

        for pred in predictions:
            matched = False
            for i, gt in enumerate(ground_truth):
                if i in matched_gt:
                    continue
                if self._is_match(pred, gt):
                    tp += 1
                    matched_gt.add(i)
                    matched = True
                    break
            if not matched:
                fp += 1

        fn = len(ground_truth) - len(matched_gt)

        precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0

        return {
            "precision": precision,
            "recall": recall,
            "f1": f1,
            "tp": tp,
            "fp": fp,
            "fn": fn,
            "total_predicted": len(predictions),
            "total_actual": len(ground_truth),
        }

    def _is_match(self, pred: Finding, gt: Finding) -> bool:
        """判断预测结果是否与真值匹配"""
        # 位置匹配
        if pred.file_path != gt.file_path:
            return False
        if abs(pred.start_line - gt.start_line) > 5:
            return False
        # CWE匹配
        if pred.cwe_id != gt.cwe_id:
            return False
        return True
```
